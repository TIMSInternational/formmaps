# Vocational 360 — Phase 1 (Instrument Model + Seed) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Model the "Evaluación 360 de Orientación Vocacional" instrument as new versioned tables, seed it from a committed data file derived from the source spreadsheet, and expose two read endpoints — without touching the live generic-360 engine.

**Architecture:** Four new Prisma models (`VocationalInstrument` / `VocationalDimension` / `VocationalQuestion` / `VocationalQuestionVariant`), global config (no tenant RLS, like `questions_360`). A one-time extraction normalizes the spreadsheet into `api/scripts/data/vocational-360-instrument.json`; an idempotent seed loads it. A thin read service + route exposes the instrument and a per-group questionnaire. Spec: `docs/superpowers/specs/2026-06-24-vocational-360-p1-design.md`.

**Tech Stack:** Prisma + Postgres, Express + zod (api), vitest + supertest (api tests), Python/openpyxl (one-time extraction).

## Global Constraints

- Model conventions: `id String @id @default(uuid())`, audit cols `isActive Boolean @default(true)`, `createdBy String?`, `createdDate DateTime @default(now())`, `updatedBy String?`, `updatedAt DateTime @updatedAt`; `@@map("snake_case")`; every FK `@@index`. Plain `String` for type/block/group/status (no Prisma enums — matches `Question360`).
- API: `res.json({ success: true, data })` / `res.status(n).json({ success: false, message })`; every catch `logger.error(err, "...")` + generic `"Internal server error"`; no `req.body` spread; no new `any` in non-test api code.
- Validated instrument shape (assertions every task implicitly upholds): **8 dimensions**, weights `[15,20,10,10,10,10,15,10]` summing to **100**, each dimension `scaleAnchors` length **5**; **45 shared questions** × 4 variants + **20 group-specific** question rows × 1 variant = **65 questions / 200 variants**; a single group's questionnaire = **50 questions** (45 shared + 5 group-specific).
- Groups (exact strings): `self`, `parent`, `teacher`, `sibling_friend`. Question types: `likert`, `ranking`, `multi_select`, `single_select`, `open`. Blocks: `dimension`, `prioritization`, `open`, `group_specific`.
- Dimension `key` slugs strip Spanish accents (`habilidades académicas → habilidades_academicas`).
- Dev DB schema apply: `cd api && npx prisma db push --accept-data-loss` (dev only). After schema edits: `npx prisma format && npx prisma generate`.
- Branch `feat/vocational-360` (already created off develop). Commit per task. No change to `Question360`/`EvaluationGroup`/`EvaluationFeedback`/take-flow/career-matching.

```prisma
// =============================================================================
// VOCATIONAL 360 (Evaluación 360 de Orientación Vocacional) — versioned instrument
// Global config (no tenant RLS), independent of the generic Question360 engine.
// =============================================================================

model VocationalInstrument {
  id                  String  @id @default(uuid())
  version             String  @unique
  name                String
  status              String  @default("draft") // draft | active
  groupWeights        Json    // { self, parent, teacher, sibling_friend }
  integrationWeights  Json    // { threeSixty, pca, mil }
  interpretationBands Json    // { strong, moderateHigh, medium }
  isActive            Boolean  @default(true)
  createdBy           String?
  createdDate         DateTime @default(now())
  updatedBy           String?
  updatedAt           DateTime @updatedAt

  dimensions VocationalDimension[]
  questions  VocationalQuestion[]

  @@map("vocational_instruments")
}

model VocationalDimension {
  id           String  @id @default(uuid())
  instrumentId String
  key          String
  nameEs       String
  nameEn       String?
  objective    String
  weight       Decimal @default(0)
  scaleAnchors Json    // array of exactly 5 strings (index 0 = score 1 … index 4 = score 5)
  order        Int     @default(0)
  isActive     Boolean  @default(true)
  createdBy    String?
  createdDate  DateTime @default(now())
  updatedBy    String?
  updatedAt    DateTime @updatedAt

  instrument VocationalInstrument @relation(fields: [instrumentId], references: [id], onDelete: Cascade)
  questions  VocationalQuestion[]

  @@unique([instrumentId, key])
  @@index([instrumentId])
  @@map("vocational_dimensions")
}

model VocationalQuestion {
  id           String  @id @default(uuid())
  instrumentId String
  dimensionId  String?
  block        String  // dimension | prioritization | open | group_specific
  number       Int
  type         String  // likert | ranking | multi_select | single_select | open
  area         String?
  scaleAnchors Json?   // null = inherit dimension's scale
  options      Json?   // [{ value, labelEs, labelEn? }] for ranking/select
  scoringRule  Json?   // { kind, ... } or null
  group        String? // set to a single group for group_specific; null = all groups
  order        Int     @default(0)
  isActive     Boolean  @default(true)
  createdBy    String?
  createdDate  DateTime @default(now())
  updatedBy    String?
  updatedAt    DateTime @updatedAt

  instrument VocationalInstrument        @relation(fields: [instrumentId], references: [id], onDelete: Cascade)
  dimension  VocationalDimension?        @relation(fields: [dimensionId], references: [id], onDelete: SetNull)
  variants   VocationalQuestionVariant[]

  @@index([instrumentId])
  @@index([dimensionId])
  @@map("vocational_questions")
}

model VocationalQuestionVariant {
  id          String  @id @default(uuid())
  questionId  String
  group       String  // self | parent | teacher | sibling_friend
  textEs      String
  textEn      String?
  isActive    Boolean  @default(true)
  createdBy   String?
  createdDate DateTime @default(now())
  updatedBy   String?
  updatedAt   DateTime @updatedAt

  question VocationalQuestion @relation(fields: [questionId], references: [id], onDelete: Cascade)

  @@unique([questionId, group])
  @@index([questionId])
  @@map("vocational_question_variants")
}
```

- [ ] **Step 1: Write the extraction script** `api/scripts/extract-vocational-360.py`:

```python
import openpyxl, json, re, unicodedata, sys, os
SRC = os.environ.get("VOC360_XLSX", os.path.expanduser(
    "~/Downloads/Evaluacion 360 Vocacional Grupos Evaluadores 2 Version (1).xlsx"))
OUT = os.path.join(os.path.dirname(__file__), "data", "vocational-360-instrument.json")
GROUPS = {"Autoevaluación": "self", "Padres": "parent", "Profesores": "teacher", "Hermanos-Amigos": "sibling_friend"}
SHARED_MAX = 45

def slug(s):
    s = unicodedata.normalize("NFKD", s).encode("ascii", "ignore").decode()
    return re.sub(r"[^a-z0-9]+", "_", s.lower()).strip("_")

def cell(ws, col, r):
    v = ws[f"{col}{r}"].value
    return v.strip() if isinstance(v, str) else v

def parse_scale(s):
    if not isinstance(s, str) or "=" not in s:
        return None
    anchors = []
    for part in re.split(r"\s*\|\s*", s):
        m = re.match(r"\s*([1-5])\s*=\s*(.+)", part)
        if m:
            anchors.append(m.group(2).strip())
    return anchors if len(anchors) == 5 else None

TYPE_MAP = [("escala 1 a 5", "likert"), ("ranking", "ranking"),
            ("selección múltiple", "multi_select"), ("selección única", "single_select"),
            ("respuesta abierta", "open"), ("escala", "likert"), ("selección", "single_select")]
def norm_type(e):
    if not isinstance(e, str):
        return "likert"
    el = e.lower()
    for k, v in TYPE_MAP:
        if k in el:
            return v
    return "open"

def scoring_rule(t):
    if t == "ranking":
        return {"kind": "rank", "topPoints": 20}
    if t == "multi_select":
        return {"kind": "pickN", "n": 5, "pointsEach": 1}
    if t == "likert":
        return {"kind": "weightedLikert"}
    return None

def split_prompt_options(text, t):
    # ranking/select questions embed a ';'-separated list after the first ':'
    if t in ("ranking", "multi_select", "single_select") and isinstance(text, str) and ":" in text:
        prompt, rest = text.split(":", 1)
        opts = [o.strip() for o in rest.split(";") if o.strip()]
        if len(opts) >= 2:
            return prompt.strip() + ":", [{"value": slug(o)[:60] or str(i), "labelEs": o} for i, o in enumerate(opts)]
    return text, None

wb = openpyxl.load_workbook(SRC, data_only=True)
data = {g: {} for g in GROUPS.values()}
for sheet, grp in GROUPS.items():
    ws = wb[sheet]
    for r in range(6, ws.max_row + 1):
        n = cell(ws, "A", r)
        if n is None:
            continue
        data[grp][n] = {k: cell(ws, c, r) for k, c in
                        {"block": "B", "dim": "C", "obj": "D", "tipo": "E", "scale": "F",
                         "q": "G", "area": "H", "pond": "I"}.items()}
nums = sorted(set().union(*[set(d) for d in data.values()]))

dims = {}
for n, row in data["self"].items():
    if row["block"] == "Preguntas por dimensión":
        d = row["dim"]
        if d not in dims:
            w = row["pond"]
            weight = float(str(w).replace("%", "").strip()) if w else 0.0
            dims[d] = {"key": slug(d), "nameEs": d, "objective": row["obj"],
                       "weight": weight, "scaleAnchors": parse_scale(row["scale"]), "order": len(dims) + 1}

questions = []
for n in nums:
    present = {g: data[g][n] for g in data if n in data[g]}
    base = present.get("self") or next(iter(present.values()))
    is_dim = base["block"] == "Preguntas por dimensión"
    t = norm_type(base["tipo"])
    if n <= SHARED_MAX:
        prompt0, opts0 = split_prompt_options(base["q"], t)
        variants = {}
        for g, row in present.items():
            p, _ = split_prompt_options(row["q"], t)
            variants[g] = {"textEs": p}
        questions.append({
            "number": n, "block": ("dimension" if is_dim else base["block"].lower()),
            "dimensionKey": dims[base["dim"]]["key"] if is_dim else None, "type": t,
            "area": base["area"], "scaleAnchors": None if is_dim else parse_scale(base["scale"]),
            "options": opts0, "scoringRule": scoring_rule(t), "group": None, "order": n,
            "variants": variants})
    else:
        for g, row in present.items():
            gt = norm_type(row["tipo"])
            p, opts = split_prompt_options(row["q"], gt)
            questions.append({
                "number": n, "block": "group_specific", "dimensionKey": None, "type": gt,
                "area": row["area"], "scaleAnchors": parse_scale(row["scale"]), "options": opts,
                "scoringRule": scoring_rule(gt), "group": g, "order": n,
                "variants": {g: {"textEs": p}}})

out = {"version": "v1", "name": "Evaluación 360 de Orientación Vocacional",
       "groupWeights": {"self": 35, "parent": 25, "teacher": 25, "sibling_friend": 15},
       "integrationWeights": {"threeSixty": 40, "pca": 30, "mil": 30},
       "interpretationBands": {"strong": 80, "moderateHigh": 60, "medium": 40},
       "dimensions": list(dims.values()), "questions": questions}

os.makedirs(os.path.dirname(OUT), exist_ok=True)
json.dump(out, open(OUT, "w"), ensure_ascii=False, indent=2)

shared = [q for q in questions if q["group"] is None]
gs = [q for q in questions if q["group"] is not None]
assert len(dims) == 8, len(dims)
assert abs(sum(d["weight"] for d in dims.values()) - 100) < 0.01
assert all(d["scaleAnchors"] and len(d["scaleAnchors"]) == 5 for d in dims.values())
assert len(shared) == 45 and all(len(q["variants"]) == 4 for q in shared)
assert len(gs) == 20 and all(len(q["variants"]) == 1 for q in gs)
print(f"OK dims={len(dims)} questions={len(questions)} shared={len(shared)} group_specific={len(gs)} variants={45*4+20}")
```

- [ ] **Step 3: Write `api/src/services/vocational360Seed.ts`:**

```ts
export interface VariantJson { textEs: string; textEn?: string | null }
export interface DimensionJson {
  key: string; nameEs: string; nameEn?: string | null; objective: string;
  weight: number; scaleAnchors: string[]; order: number;
}
export interface QuestionJson {
  number: number; block: string; dimensionKey: string | null; type: string;
  area: string | null; scaleAnchors: string[] | null; options: unknown | null;
  scoringRule: unknown | null; group: string | null; order: number;
  variants: Record<string, VariantJson>;
}
export interface InstrumentJson {
  version: string; name: string; groupWeights: unknown; integrationWeights: unknown;
  interpretationBands: unknown; dimensions: DimensionJson[]; questions: QuestionJson[];
}

export interface SeedPayload {
  instrument: { version: string; name: string; groupWeights: unknown; integrationWeights: unknown; interpretationBands: unknown };
  dimensions: DimensionJson[];
  questions: QuestionJson[];
}

/** Pure transform + invariant guard. Throws on a malformed instrument so a bad
 *  data file never half-seeds the DB. */
export function buildSeedPayload(json: InstrumentJson): SeedPayload {
  if (json.dimensions.length !== 8) throw new Error(`Expected 8 dimensions, got ${json.dimensions.length}`);
  const weightSum = json.dimensions.reduce((s, d) => s + d.weight, 0);
  if (Math.round(weightSum) !== 100) throw new Error(`Dimension weights must sum to 100, got ${weightSum}`);
  for (const d of json.dimensions) {
    if (!Array.isArray(d.scaleAnchors) || d.scaleAnchors.length !== 5) {
      throw new Error(`Dimension ${d.key} must have 5 scale anchors`);
    }
  }
  for (const q of json.questions) {
    const variantGroups = Object.keys(q.variants);
    if (q.group === null && variantGroups.length !== 4) {
      throw new Error(`Shared question ${q.number} must have 4 variants, got ${variantGroups.length}`);
    }
    if (q.group !== null && variantGroups.length !== 1) {
      throw new Error(`Group-specific question ${q.number}/${q.group} must have 1 variant`);
    }
  }
  return {
    instrument: {
      version: json.version, name: json.name, groupWeights: json.groupWeights,
      integrationWeights: json.integrationWeights, interpretationBands: json.interpretationBands,
    },
    dimensions: json.dimensions,
    questions: json.questions,
  };
}
```

- [ ] **Step 5: Write the idempotent seed script** `api/scripts/seed-vocational-360.ts`:

```ts
import "dotenv/config";
import { PrismaClient, Prisma } from "@prisma/client";
import fs from "fs";
import path from "path";
import { buildSeedPayload, InstrumentJson } from "../src/services/vocational360Seed.js";

const prisma = new PrismaClient();

async function main() {
  const json: InstrumentJson = JSON.parse(
    fs.readFileSync(path.join(__dirname, "data", "vocational-360-instrument.json"), "utf8"),
  );
  const payload = buildSeedPayload(json);

  const instrument = await prisma.vocationalInstrument.upsert({
    where: { version: payload.instrument.version },
    update: {
      name: payload.instrument.name,
      groupWeights: payload.instrument.groupWeights as Prisma.InputJsonValue,
      integrationWeights: payload.instrument.integrationWeights as Prisma.InputJsonValue,
      interpretationBands: payload.instrument.interpretationBands as Prisma.InputJsonValue,
      status: "active",
    },
    create: {
      version: payload.instrument.version, name: payload.instrument.name, status: "active",
      groupWeights: payload.instrument.groupWeights as Prisma.InputJsonValue,
      integrationWeights: payload.instrument.integrationWeights as Prisma.InputJsonValue,
      interpretationBands: payload.instrument.interpretationBands as Prisma.InputJsonValue,
    },
  });

  const dimIdByKey = new Map<string, string>();
  for (const d of payload.dimensions) {
    const row = await prisma.vocationalDimension.upsert({
      where: { instrumentId_key: { instrumentId: instrument.id, key: d.key } },
      update: { nameEs: d.nameEs, nameEn: d.nameEn ?? null, objective: d.objective,
        weight: new Prisma.Decimal(d.weight), scaleAnchors: d.scaleAnchors as Prisma.InputJsonValue, order: d.order },
      create: { instrumentId: instrument.id, key: d.key, nameEs: d.nameEs, nameEn: d.nameEn ?? null,
        objective: d.objective, weight: new Prisma.Decimal(d.weight),
        scaleAnchors: d.scaleAnchors as Prisma.InputJsonValue, order: d.order },
    });
    dimIdByKey.set(d.key, row.id);
  }

  for (const q of payload.questions) {
    // natural key: (instrumentId, number, group). group null encoded as "__all__" for the find.
    const existing = await prisma.vocationalQuestion.findFirst({
      where: { instrumentId: instrument.id, number: q.number, group: q.group },
    });
    const qData = {
      instrumentId: instrument.id, dimensionId: q.dimensionKey ? dimIdByKey.get(q.dimensionKey)! : null,
      block: q.block, number: q.number, type: q.type, area: q.area ?? null,
      scaleAnchors: (q.scaleAnchors ?? undefined) as Prisma.InputJsonValue | undefined,
      options: (q.options ?? undefined) as Prisma.InputJsonValue | undefined,
      scoringRule: (q.scoringRule ?? undefined) as Prisma.InputJsonValue | undefined,
      group: q.group, order: q.order,
    };
    const question = existing
      ? await prisma.vocationalQuestion.update({ where: { id: existing.id }, data: qData })
      : await prisma.vocationalQuestion.create({ data: qData });

    for (const [group, v] of Object.entries(q.variants)) {
      await prisma.vocationalQuestionVariant.upsert({
        where: { questionId_group: { questionId: question.id, group } },
        update: { textEs: v.textEs, textEn: v.textEn ?? null },
        create: { questionId: question.id, group, textEs: v.textEs, textEn: v.textEn ?? null },
      });
    }
  }

  const counts = {
    instruments: await prisma.vocationalInstrument.count(),
    dimensions: await prisma.vocationalDimension.count({ where: { instrumentId: instrument.id } }),
    questions: await prisma.vocationalQuestion.count({ where: { instrumentId: instrument.id } }),
    variants: await prisma.vocationalQuestionVariant.count({ where: { question: { instrumentId: instrument.id } } }),
  };
  console.log("Vocational 360 seeded:", counts);
  if (counts.dimensions !== 8 || counts.questions !== 65 || counts.variants !== 200) {
    throw new Error("Seed counts off: " + JSON.stringify(counts));
  }
}
```

```ts
import { prisma } from "../lib/prisma.js";

export const VALID_GROUPS = ["self", "parent", "teacher", "sibling_friend"] as const;
export type VocationalGroup = (typeof VALID_GROUPS)[number];

export interface InstrumentDimensionDto {
  key: string; nameEs: string; nameEn: string | null; weight: number; scaleAnchors: string[]; order: number;
}
export interface InstrumentDto {
  version: string; name: string; groupWeights: unknown; integrationWeights: unknown;
  interpretationBands: unknown; dimensions: InstrumentDimensionDto[];
}
export interface QuestionnaireItem {
  number: number; block: string; type: string; area: string | null; dimensionKey: string | null;
  scaleAnchors: string[] | null; options: unknown | null; text: string;
}

export async function getInstrument(): Promise<InstrumentDto | null> {
  const inst = await prisma.vocationalInstrument.findFirst({
    where: { status: "active", isActive: true },
    include: { dimensions: { where: { isActive: true }, orderBy: { order: "asc" } } },
  });
  if (!inst) return null;
  return {
    version: inst.version, name: inst.name, groupWeights: inst.groupWeights,
    integrationWeights: inst.integrationWeights, interpretationBands: inst.interpretationBands,
    dimensions: inst.dimensions.map((d) => ({
      key: d.key, nameEs: d.nameEs, nameEn: d.nameEn, weight: Number(d.weight),
      scaleAnchors: d.scaleAnchors as string[], order: d.order,
    })),
  };
}

export async function getQuestionnaire(group: string): Promise<QuestionnaireItem[]> {
  if (!VALID_GROUPS.includes(group as VocationalGroup)) {
    throw new Error(`Invalid group: ${group}`);
  }
  const questions = await prisma.vocationalQuestion.findMany({
    where: { isActive: true, OR: [{ group: null }, { group }] },
    orderBy: { order: "asc" },
    include: {
      dimension: { select: { key: true, scaleAnchors: true } },
      variants: { where: { group, isActive: true } },
    },
  });
  return questions.map((q) => ({
    number: q.number, block: q.block, type: q.type, area: q.area,
    dimensionKey: q.dimension?.key ?? null,
    scaleAnchors: ((q.scaleAnchors as string[] | null) ?? (q.dimension?.scaleAnchors as string[] | undefined)) ?? null,
    options: q.options ?? null,
    text: q.variants[0]?.textEs ?? "",
  }));
}
```

```ts
import { Router, Request, Response } from "express";
import { logger } from "../lib/logger.js";
import { qs } from "../lib/query.js";
import { getInstrument, getQuestionnaire } from "../services/vocational360Service.js";

const router = Router();

router.get("/instrument", async (_req: Request, res: Response) => {
  try {
    const data = await getInstrument();
    if (!data) {
      res.status(404).json({ success: false, message: "No active vocational instrument" });
      return;
    }
    res.json({ success: true, data });
  } catch (err) {
    logger.error(err, "Vocational instrument fetch error:");
    res.status(500).json({ success: false, message: "Internal server error" });
  }
});

router.get("/questionnaire", async (req: Request, res: Response) => {
  try {
    const group = (qs(req.query.group) || "").trim();
    const data = await getQuestionnaire(group);
    res.json({ success: true, data });
  } catch (err) {
    if (err instanceof Error && /invalid group/i.test(err.message)) {
      res.status(400).json({ success: false, message: "Invalid group" });
      return;
    }
    logger.error(err, "Vocational questionnaire fetch error:");
    res.status(500).json({ success: false, message: "Internal server error" });
  }
});

export default router;
```
