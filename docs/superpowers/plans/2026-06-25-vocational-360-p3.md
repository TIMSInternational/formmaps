# Vocational 360 — Phase 3 (Scoring Engine) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Turn a student's completed vocational evaluations into a persisted, versioned 360 result — per-dimension scores, a group-weighted composite + interpretation bands, and interest/industry/work-type rankings — with read + recompute endpoints and an auto-recompute hook.

**Architecture:** A pure, fixture-testable scoring core (`vocationalScoringService.ts`) does all math; a thin persistence/orchestration layer (`vocational360Service.ts`, extended) loads Prisma data → computes → upserts a new `vocational_results` snapshot; two school-scoped endpoints read/recompute; a guarded best-effort hook in the P2a submit path keeps snapshots fresh. Backend-only. PCA/MIL integration + report UI are P4.

**Tech Stack:** Node + Express + Prisma (PostgreSQL), vitest (mocked Prisma), TypeScript. Spec: `docs/superpowers/specs/2026-06-25-vocational-360-p3-design.md`.

## Global Constraints

- API response format ALWAYS `res.json({ success: true, data })` / `res.status(n).json({ success: false, message })`.
- Service-layer pattern: Route (validate + auth) → Service (logic) → Prisma. No `req.body` spread into Prisma; pick fields explicitly.
- **No new `any`** in non-test code — use `unknown` + narrow. (`as`-casts allowed in test files.)
- IDOR: every `:id` route verifies access via `canAccessUser({ role: req.userRole || "", userId: req.userId!, schoolId: req.schoolId }, targetUserId)`; on deny return **404** (not 403). Never expose `err.message` to clients.
- Decimal columns serialize to strings over JSON → convert Decimal→Number at the route boundary (`serializeVocationalResult`).
- **ISOLATION:** do NOT change behavior of the generic 360 engine (`Question360`/`EvaluationFeedback`/`submit-feedback`/`averageRating`), the assessment-completion gate, or the P2 vocational take/submit flow — except the one ADDITIVE, error-swallowed recompute hook appended at the end of `submitVocational` (Task 6). No existing table is modified; the only schema change is the NEW `vocational_results` table.
- No prod DDL in this branch (dev `prisma db push` only; prod table ships later via Fargate idempotent CREATE).
- Branch `feat/vocational-360-p3` (off develop, has P1+P2a+P2b). Backend test cmds run from `api/`. Commit per task.

## Group + scoring facts (from P1 seed / instrument JSON)
- Groups: `self | parent | teacher | sibling_friend`. Group weights `{self:35, parent:25, teacher:25, sibling_friend:15}`. Bands `{strong:80, moderateHigh:60, medium:40}` (below 40 → `low`).
- 8 dimensions, weights `[15,20,10,10,10,10,15,10]`. Normalization `((r−1)/4)×100`.
- Per-question `scoringRule.kind`: `weightedLikert` (dimension Likerts), `rank` (`topPoints`), `pickN` (`n`,`pointsEach`), or `null` (single_select work-type + open). `single_select`/`open` never feed the dimension composite.

export const VOCATIONAL_GROUPS = ["self", "parent", "teacher", "sibling_friend"] as const;
export type VocGroup = (typeof VOCATIONAL_GROUPS)[number];

export interface ScoringResponse {
  questionNumber: number;
  type: string; // likert | ranking | multi_select | single_select | open
  dimensionKey: string | null;
  ratingValue: number | null;
  rankingOrder: { value: string; rank: number }[] | null;
  selectedValues: string[] | null;
  textValue: string | null;
}
export interface ScoringGroup { group: VocGroup; responses: ScoringResponse[] }
export interface ScoringDimension { key: string; nameEs: string; weight: number }
export interface ScoringBands { strong: number; moderateHigh: number; medium: number }
export interface ScoringQuestion {
  number: number;
  type: string;
  scoringRule: { kind: string; topPoints?: number; n?: number; pointsEach?: number } | null;
}
export interface ScoringConfig {
  instrumentVersion: string;
  groupWeights: Record<VocGroup, number>;
  bands: ScoringBands;
  dimensions: ScoringDimension[];
}
export interface DimensionScore {
  key: string;
  nameEs: string;
  score: number | null;
  band: string | null;
  byGroup: Partial<Record<VocGroup, number>>;
}
export interface Rankings {
  interests: { value: string; points: number }[];
  industries: { value: string; count: number }[];
  workType: { value: string; count: number } | null;
  openInsights: { group: string; text: string }[];
}
export interface VocationalResultPayload {
  status: "ready";
  instrumentVersion: string;
  composite: number;
  band: string;
  respondentCount: number;
  groupsIncluded: VocGroup[];
  dimensionScores: DimensionScore[];
  rankings: Rankings;
  weightsApplied: Partial<Record<VocGroup, number>>;
}
export type ScoringOutcome = VocationalResultPayload | { status: "not_ready"; reason: string };

export function round2(n: number): number {
  return Math.round(n * 100) / 100;
}

export function normalize(rating: number): number {
  return ((rating - 1) / 4) * 100;
}

export function band(score: number, b: ScoringBands): string {
  if (score >= b.strong) return "strong";
  if (score >= b.moderateHigh) return "moderateHigh";
  if (score >= b.medium) return "medium";
  return "low";
}

export function dimensionScoreForGroup(responses: ScoringResponse[], dimensionKey: string): number | null {
  const vals = responses
    .filter((r) => r.type === "likert" && r.dimensionKey === dimensionKey && typeof r.ratingValue === "number")
    .map((r) => normalize(r.ratingValue as number));
  if (vals.length === 0) return null;
  return vals.reduce((a, b) => a + b, 0) / vals.length;
}

export function renormalizeGroupWeights(
  base: Record<VocGroup, number>,
  present: VocGroup[],
): Partial<Record<VocGroup, number>> {
  const out: Partial<Record<VocGroup, number>> = {};
  const sum = present.reduce((s, g) => s + (base[g] ?? 0), 0);
  if (sum === 0) {
    const eq = present.length ? 1 / present.length : 0;
    for (const g of present) out[g] = eq;
    return out;
  }
  for (const g of present) out[g] = (base[g] ?? 0) / sum;
  return out;
}

export function aggregateDimension(
  byGroup: Partial<Record<VocGroup, number>>,
  base: Record<VocGroup, number>,
): number | null {
  const present = (Object.keys(byGroup) as VocGroup[]).filter((g) => typeof byGroup[g] === "number");
  if (present.length === 0) return null;
  const w = renormalizeGroupWeights(base, present);
  return present.reduce((s, g) => s + (byGroup[g] as number) * (w[g] as number), 0);
}

export function composite(dims: DimensionScore[], dimMeta: ScoringDimension[]): number {
  const wByKey = new Map(dimMeta.map((d) => [d.key, d.weight]));
  const scored = dims.filter((d) => typeof d.score === "number");
  const wsum = scored.reduce((s, d) => s + (wByKey.get(d.key) ?? 0), 0);
  if (wsum === 0) return 0;
  return scored.reduce((s, d) => s + (d.score as number) * ((wByKey.get(d.key) ?? 0) / wsum), 0);
}

```ts
export function computeRankings(
  groups: ScoringGroup[],
  base: Record<VocGroup, number>,
  questions: ScoringQuestion[],
): Rankings {
  const present = groups.map((g) => g.group);
  const w = renormalizeGroupWeights(base, present);
  const qByNum = new Map(questions.map((q) => [q.number, q]));
  const interestPts = new Map<string, number>();
  const industryCnt = new Map<string, number>();
  const workTypeCnt = new Map<string, number>();
  const openInsights: { group: string; text: string }[] = [];

  for (const grp of groups) {
    const gw = w[grp.group] ?? 0;
    for (const r of grp.responses) {
      if (r.type === "ranking" && r.rankingOrder) {
        const topPoints = qByNum.get(r.questionNumber)?.scoringRule?.topPoints ?? 20;
        for (const { value, rank } of r.rankingOrder) {
          const pts = Math.max(0, topPoints - (rank - 1)) * gw;
          interestPts.set(value, (interestPts.get(value) ?? 0) + pts);
        }
      } else if (r.type === "multi_select" && r.selectedValues) {
        for (const v of r.selectedValues) industryCnt.set(v, (industryCnt.get(v) ?? 0) + gw);
      } else if (r.type === "single_select" && r.textValue) {
        workTypeCnt.set(r.textValue, (workTypeCnt.get(r.textValue) ?? 0) + gw);
      } else if (r.type === "open" && r.textValue && r.textValue.trim()) {
        openInsights.push({ group: grp.group, text: r.textValue.trim() });
      }
    }
  }

  const interests = [...interestPts.entries()]
    .map(([value, points]) => ({ value, points: round2(points) }))
    .sort((a, b) => b.points - a.points);
  const industries = [...industryCnt.entries()]
    .map(([value, count]) => ({ value, count: round2(count) }))
    .sort((a, b) => b.count - a.count);
  let workType: { value: string; count: number } | null = null;
  if (workTypeCnt.size) {
    const top = [...workTypeCnt.entries()].sort((a, b) => b[1] - a[1])[0];
    workType = { value: top[0], count: round2(top[1]) };
  }
  return { interests, industries, workType, openInsights };
}

export function computeVocationalResult(
  config: ScoringConfig,
  questions: ScoringQuestion[],
  groups: ScoringGroup[],
): ScoringOutcome {
  const present = groups.map((g) => g.group);
  const hasSelf = present.includes("self");
  const others = present.filter((g) => g !== "self");
  if (!hasSelf || others.length === 0) {
    return { status: "not_ready", reason: "needs_self_plus_one" };
  }

  const dimensionScores: DimensionScore[] = config.dimensions.map((d) => {
    const byGroup: Partial<Record<VocGroup, number>> = {};
    for (const grp of groups) {
      const s = dimensionScoreForGroup(grp.responses, d.key);
      if (s !== null) byGroup[grp.group] = round2(s);
    }
    const agg = aggregateDimension(byGroup, config.groupWeights);
    return {
      key: d.key,
      nameEs: d.nameEs,
      score: agg === null ? null : round2(agg),
      band: agg === null ? null : band(agg, config.bands),
      byGroup,
    };
  });

  const comp = round2(composite(dimensionScores, config.dimensions));
  return {
    status: "ready",
    instrumentVersion: config.instrumentVersion,
    composite: comp,
    band: band(comp, config.bands),
    respondentCount: groups.length,
    groupsIncluded: present,
    dimensionScores,
    rankings: computeRankings(groups, config.groupWeights, questions),
    weightsApplied: renormalizeGroupWeights(config.groupWeights, present),
  };
}
```

function asNumberRecord(v: unknown): Record<VocGroup, number> {
  const o = (v ?? {}) as Record<string, unknown>;
  const out = {} as Record<VocGroup, number>;
  for (const g of VOCATIONAL_GROUPS) out[g] = typeof o[g] === "number" ? (o[g] as number) : 0;
  return out;
}

function asBands(v: unknown): ScoringConfig["bands"] {
  const o = (v ?? {}) as Record<string, unknown>;
  return {
    strong: typeof o.strong === "number" ? o.strong : 80,
    moderateHigh: typeof o.moderateHigh === "number" ? o.moderateHigh : 60,
    medium: typeof o.medium === "number" ? o.medium : 40,
  };
}

interface RawResponse {
  questionNumber: number; type: string; dimensionKey: string | null;
  ratingValue: number | null; rankingOrder: unknown; selectedValues: unknown; textValue: string | null;
}

function toScoringResponse(r: RawResponse): ScoringResponse {
  return {
    questionNumber: r.questionNumber,
    type: r.type,
    dimensionKey: r.dimensionKey,
    ratingValue: r.ratingValue,
    rankingOrder: Array.isArray(r.rankingOrder) ? (r.rankingOrder as { value: string; rank: number }[]) : null,
    selectedValues: Array.isArray(r.selectedValues) ? (r.selectedValues as string[]) : null,
    textValue: r.textValue,
  };
}

async function loadScoringInputs(evaluatedUserId: string): Promise<
  | { ok: false }
  | { ok: true; config: ScoringConfig; questions: ScoringQuestion[]; groups: ScoringGroup[] }
> {
  const inst = await prisma.vocationalInstrument.findFirst({
    where: { status: "active", isActive: true },
    include: { dimensions: { where: { isActive: true }, orderBy: { order: "asc" } } },
  });
  if (!inst) return { ok: false };

  const config: ScoringConfig = {
    instrumentVersion: inst.version,
    groupWeights: asNumberRecord(inst.groupWeights),
    bands: asBands(inst.interpretationBands),
    dimensions: inst.dimensions.map((d) => ({ key: d.key, nameEs: d.nameEs, weight: Number(d.weight) })),
  };

  const qs = await prisma.vocationalQuestion.findMany({
    where: { isActive: true },
    select: { number: true, type: true, scoringRule: true },
  });
  const questions: ScoringQuestion[] = qs.map((q) => ({
    number: q.number,
    type: q.type,
    scoringRule: (q.scoringRule as ScoringQuestion["scoringRule"]) ?? null,
  }));

  const rawGroups = await prisma.evaluationGroup.findMany({
    where: { evaluatedUserId, instrument: "vocational", isEvaluationCompleted: true, isActive: true },
    include: { vocationalResponses: { where: { isActive: true } } },
  });
  const groups: ScoringGroup[] = rawGroups
    .filter((g) => (VOCATIONAL_GROUPS as readonly string[]).includes(g.groupType))
    .map((g) => ({
      group: g.groupType as VocGroup,
      responses: g.vocationalResponses.map((r) => toScoringResponse(r as unknown as RawResponse)),
    }));

  return { ok: true, config, questions, groups };
}

export async function recomputeVocationalResult(
  evaluatedUserId: string,
): Promise<ScoringOutcome | { status: "never_computed" }> {
  const inputs = await loadScoringInputs(evaluatedUserId);
  if (!inputs.ok) return { status: "never_computed" };

  const outcome = computeVocationalResult(inputs.config, inputs.questions, inputs.groups);
  if (outcome.status !== "ready") return outcome;

  await prisma.vocationalResult.upsert({
    where: { evaluatedUserId_instrumentVersion: { evaluatedUserId, instrumentVersion: outcome.instrumentVersion } },
    update: {
      composite: outcome.composite, band: outcome.band, respondentCount: outcome.respondentCount,
      groupsIncluded: outcome.groupsIncluded, dimensionScores: outcome.dimensionScores as object,
      rankings: outcome.rankings as object, weightsApplied: outcome.weightsApplied as object,
    },
    create: {
      evaluatedUserId, instrumentVersion: outcome.instrumentVersion,
      composite: outcome.composite, band: outcome.band, respondentCount: outcome.respondentCount,
      groupsIncluded: outcome.groupsIncluded, dimensionScores: outcome.dimensionScores as object,
      rankings: outcome.rankings as object, weightsApplied: outcome.weightsApplied as object,
    },
  });
  return outcome;
}

export interface SerializedResult {
  status: "ready";
  evaluatedUserId: string;
  instrumentVersion: string;
  composite: number;
  band: string;
  respondentCount: number;
  groupsIncluded: string[];
  dimensionScores: unknown;
  rankings: unknown;
  weightsApplied: unknown;
  computedAt: Date;
}

export function serializeVocationalResult(row: {
  evaluatedUserId: string; instrumentVersion: string; composite: unknown; band: string;
  respondentCount: number; groupsIncluded: string[]; dimensionScores: unknown; rankings: unknown;
  weightsApplied: unknown; computedAt: Date;
}): SerializedResult {
  return {
    status: "ready",
    evaluatedUserId: row.evaluatedUserId,
    instrumentVersion: row.instrumentVersion,
    composite: Number(row.composite),
    band: row.band,
    respondentCount: row.respondentCount,
    groupsIncluded: row.groupsIncluded,
    dimensionScores: row.dimensionScores,
    rankings: row.rankings,
    weightsApplied: row.weightsApplied,
    computedAt: row.computedAt,
  };
}

export async function getVocationalResult(
  evaluatedUserId: string,
): Promise<SerializedResult | { status: "never_computed" }> {
  const inst = await prisma.vocationalInstrument.findFirst({
    where: { status: "active", isActive: true },
    select: { version: true },
  });
  const version = inst?.version;
  const row = version
    ? await prisma.vocationalResult.findUnique({
        where: { evaluatedUserId_instrumentVersion: { evaluatedUserId, instrumentVersion: version } },
      })
    : null;
  if (!row) return { status: "never_computed" };
  return serializeVocationalResult(row);
}
