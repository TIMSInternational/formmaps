# Vocational 360 — Phase 2a (Backend Collection) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Let the 4 evaluator groups take and submit the vocational questionnaire through a new public token-keyed take+submit route storing typed `VocationalResponse` rows, reusing the EvaluationGroup invite/token/completion machinery — with the live generic-360 take-flow untouched.

**Architecture:** Additive `instrument`/`instrumentVersion` discriminator on `EvaluationGroup`; a new typed `VocationalResponse` table; a dedicated public router `vocationalTake.ts` (GET take + POST submit) mounted at `/evaluation/vocational` (before `/evaluation`), serving P1's `getQuestionnaire(group)` and validating typed answers. Spec: `docs/superpowers/specs/2026-06-24-vocational-360-p2a-design.md`.

**Tech Stack:** Prisma + Postgres, Express + zod (api), vitest + supertest.

## Global Constraints

- Model conventions: `id String @id @default(uuid())`, audit cols `isActive/createdBy/createdDate/updatedBy/updatedAt`, `@@map("snake_case")`, FK `@@index`. Plain `String` for type/group/instrument (no enums).
- API: `res.json({ success: true, data })` / `res.status(n).json({ success: false, message })`; every catch `logger.error(err, "...")` + generic `"Internal server error"`; no `req.body` spread (pick fields); bound all strings/arrays; no new `any` in non-test api code.
- Public vocational routes use `systemContext` (NOT `authenticate`); submit also uses `sensitiveLimiter`. The token in the URL/body is the only credential. 404 on bad/expired/non-vocational token (don't reveal which); 400 on validation; 409 on already-completed.
- Groups (exact): `self|parent|teacher|sibling_friend`. Reuse `normalizeGroupType` (`api/src/lib/evaluationGroups.ts`) — `other`→no questionnaire (400). Question types: `likert|ranking|multi_select|single_select|open`. Likert scale is always 1–5 (anchors are labels).
- **ISOLATION (hard requirement):** do NOT modify `submit-feedback`, `get360EvaluatorForm`, `submitFeedback`, `averageRating`, `lib/evaluation360`, or `Question360`. The only edits to live files are: 2 additive `EvaluationGroup` columns + a back-relation; a conditional-only addition to `createEvaluationGroup`'s create data; an optional `instrument`/`instrumentVersion` passthrough on the `POST /create-group` route. The generic evaluation test suite MUST stay green (isolation proof, run in the close gate).
- Dev DB apply: `cd api && npx prisma db push --accept-data-loss`. After schema edits: `npx prisma format && npx prisma generate`.
- Branch `feat/vocational-360-p2a` (already created off develop). Commit per task.

- [ ] **Step 3: Create `api/src/services/vocationalTakeService.ts`:**

```ts
import { prisma } from "../lib/prisma.js";
import { normalizeGroupType } from "../lib/evaluationGroups.js";
import { getQuestionnaire } from "./vocational360Service.js";

export type TakeResult =
  | { error: "not-found" | "expired" | "invalid-group" }
  | { completed: true; evaluatorName: string }
  | {
      group: string; instrumentVersion: string | null; evaluatorName: string;
      studentName: string | null; isEvaluationCompleted: false; questions: Awaited<ReturnType<typeof getQuestionnaire>>;
    };

/** Resolve a vocational evaluator's questionnaire by token. Mirrors get360EvaluatorForm
 *  (findFirst on invitationToken + isActive, early-return on completed) + adds an
 *  expiry guard and the instrument==="vocational" gate. */
export async function getVocationalForm(token: string): Promise<TakeResult> {
  const group = await prisma.evaluationGroup.findFirst({ where: { invitationToken: token, isActive: true } });
  if (!group || group.instrument !== "vocational") return { error: "not-found" };
  if (group.tokenExpiryDate < new Date()) return { error: "expired" };
  if (group.isEvaluationCompleted) return { completed: true, evaluatorName: group.evaluatorName };

  const g = normalizeGroupType(group.groupType);
  if (g === "other") return { error: "invalid-group" };

  const questions = await getQuestionnaire(g);
  const student = await prisma.user.findUnique({ where: { id: group.evaluatedUserId }, select: { name: true } });
  return {
    group: g, instrumentVersion: group.instrumentVersion, evaluatorName: group.evaluatorName,
    studentName: student?.name ?? null, isEvaluationCompleted: false, questions,
  };
}
```

```ts
import { systemContext } from "../middleware/tenantContext.js";
import { logger } from "../lib/logger.js";
import { qs } from "../lib/query.js";
import { getVocationalForm } from "../services/vocationalTakeService.js";

const router = Router();

// GET /evaluation/vocational/:token — public, token-keyed
router.get("/:token", systemContext, async (req: Request, res: Response) => {
  try {
    const result = await getVocationalForm(qs(req.params.token));
    if ("error" in result) {
      const status = result.error === "invalid-group" ? 400 : 404;
      const message = result.error === "invalid-group" ? "No questionnaire for this group" : "Not found";
      res.status(status).json({ success: false, message });
      return;
    }
    res.json({ success: true, data: result });
  } catch (err) {
    logger.error(err, "Vocational take error:");
    res.status(500).json({ success: false, message: "Internal server error" });
  }
});

export default router;
```

```ts
export interface VocationalAnswer {
  questionNumber: number;
  type: "likert" | "ranking" | "multi_select" | "single_select" | "open";
  ratingValue?: number;
  rankingOrder?: { value: string; rank: number }[];
  selectedValues?: string[];
  textValue?: string;
}

export type SubmitResult =
  | { ok: true; count: number }
  | { error: "not-found" | "expired" | "already-completed" | "invalid-group" | "bad-answer" };

function optionValues(options: unknown): string[] {
  return Array.isArray(options) ? (options as { value: string }[]).map((o) => o.value) : [];
}

/** Validate typed answers against the group's questionnaire, then upsert responses
 *  + flip completion in one transaction. Public path — token is the credential. */
export async function submitVocational(token: string, answers: VocationalAnswer[]): Promise<SubmitResult> {
  const group = await prisma.evaluationGroup.findFirst({ where: { invitationToken: token, isActive: true } });
  if (!group || group.instrument !== "vocational") return { error: "not-found" };
  if (group.tokenExpiryDate < new Date()) return { error: "expired" };
  if (group.isEvaluationCompleted) return { error: "already-completed" };

  const g = normalizeGroupType(group.groupType);
  if (g === "other") return { error: "invalid-group" };

  const questions = await getQuestionnaire(g);
  const byNum = new Map(questions.map((q) => [q.number, q]));

  // Validate every answer before any write.
  for (const a of answers) {
    const q = byNum.get(a.questionNumber);
    if (!q || q.type !== a.type) return { error: "bad-answer" };
    if (a.type === "ranking") {
      const opts = optionValues(q.options);
      const order = a.rankingOrder ?? [];
      const values = order.map((o) => o.value);
      const ranks = order.map((o) => o.rank);
      if (values.some((v) => !opts.includes(v))) return { error: "bad-answer" };
      if (new Set(values).size !== values.length || new Set(ranks).size !== ranks.length) return { error: "bad-answer" };
    } else if (a.type === "multi_select") {
      const opts = optionValues(q.options);
      if ((a.selectedValues ?? []).some((v) => !opts.includes(v))) return { error: "bad-answer" };
    } else if (a.type === "single_select") {
      if (!optionValues(q.options).includes(a.textValue ?? "")) return { error: "bad-answer" };
    }
    // likert range + open non-empty are enforced by the route's zod schema.
  }

  const ops = answers.map((a) => {
    const q = byNum.get(a.questionNumber)!;
    const fields = {
      instrumentVersion: group.instrumentVersion ?? "", group: g, dimensionKey: q.dimensionKey ?? null, type: a.type,
      ratingValue: a.type === "likert" ? a.ratingValue ?? null : null,
      rankingOrder: a.type === "ranking" ? (a.rankingOrder as unknown) : undefined,
      selectedValues: a.type === "multi_select" ? (a.selectedValues as unknown) : undefined,
      textValue: a.type === "single_select" || a.type === "open" ? a.textValue ?? null : null,
    };
    return prisma.vocationalResponse.upsert({
      where: { evaluationGroupId_questionNumber: { evaluationGroupId: group.id, questionNumber: a.questionNumber } },
      update: fields,
      create: { evaluationGroupId: group.id, questionNumber: a.questionNumber, ...fields } as never,
    });
  });

  await prisma.$transaction([
    ...ops,
    prisma.evaluationGroup.update({
      where: { id: group.id },
      data: { isTokenUsed: true, tokenUsedDate: new Date(), isEvaluationCompleted: true, evaluationCompletedDate: new Date() },
    }),
  ]);

  return { ok: true, count: answers.length };
}
```

```ts
import { sensitiveLimiter } from "../middleware/rateLimiter.js";

  z.object({ questionNumber: z.number().int(), type: z.literal("open"), textValue: z.string().min(1).max(4000) }),
]);
const submitSchema = z.object({ token: z.string().min(1).max(200), answers: z.array(answerSchema).min(1).max(80) });

// POST /evaluation/vocational/submit — public, token-keyed
router.post("/submit", systemContext, sensitiveLimiter, async (req: Request, res: Response) => {
  try {
    const body = submitSchema.safeParse(req.body);
    if (!body.success) {
      res.status(400).json({ success: false, message: "Invalid submission" });
      return;
    }
    const result = await submitVocational(body.data.token, body.data.answers);
    if ("error" in result) {
      const status = result.error === "already-completed" ? 409
        : result.error === "invalid-group" || result.error === "bad-answer" ? 400 : 404;
      res.status(status).json({ success: false, message: status === 409 ? "Already completed" : status === 400 ? "Invalid submission" : "Not found" });
      return;
    }
    res.json({ success: true, data: result });
  } catch (err) {
    logger.error(err, "Vocational submit error:");
    res.status(500).json({ success: false, message: "Internal server error" });
  }
});
```
