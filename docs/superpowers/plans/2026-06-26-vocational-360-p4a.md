# Vocational 360 — Phase 4a (360/PCA/MIL Integration Engine) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Produce the persisted, versioned integrated vocational score `0.4×(360 composite) + 0.3×(PCA competences avg) + 0.3×(MIL cognitive mean)` with an interpretation band, plus read/recompute endpoints.

**Architecture:** A pure, fixture-testable integration service (`vocationalIntegrationService.ts`, reusing P3's `band`/`round2`) does the math; orchestration in `vocational360Service.ts` pulls the 360 composite (P3 `getVocationalResult`) + PCA/MIL (canonical `assembleCompleteProfile`) → computes → upserts a new `vocational_integrated_results` snapshot; two school-scoped endpoints read/recompute. Backend-only; report UI = P4b, recommendations = P4c.

**Tech Stack:** Node + Express + Prisma (PostgreSQL), vitest (mocked Prisma), TypeScript ESM (`.js` import extensions). Spec: `docs/superpowers/specs/2026-06-26-vocational-360-p4a-design.md`.

## Global Constraints

- API response format ALWAYS `res.json({ success: true, data })` / `res.status(n).json({ success: false, message })`.
- Service-layer pattern (Route validate+auth → Service → Prisma); no `req.body`/object spread into Prisma — pick fields explicitly.
- **No new `any`** in non-test code — `unknown` + narrow. (`as`-casts allowed in test files, eslint-disabled.)
- IDOR: every `:id` route verifies `canAccessUser({ role: req.userRole || "", userId: req.userId!, schoolId: req.schoolId }, targetId)`; deny → **404** (not 403); never leak `err.message`; bind param `(qs(...) || "").slice(0,100)`.
- Decimal columns → Number at the route boundary (`serializeIntegratedResult`).
- **Weights from data:** integration weights + bands come from the active `VocationalInstrument` (`integrationWeights` `{threeSixty:40,pca:30,mil:30}`, `interpretationBands` `{strong:80,moderateHigh:60,medium:40}`), NOT hardcoded. Reuse `band` + `round2` from `vocationalScoringService.js` — do not re-implement.
- **Component scalarization:** PCA = `((mean competence level − 1) / 3) × 100` (levels 1–4); MIL = simple mean of the five `lia.mil` domain scores (0–100). 360 = P3 `getVocationalResult` composite when `status==="ready"`.
- **Guard:** require all three (360 ready + PCA competences present + MIL `completeness.lia`); else `not_ready` with `missing:[...]`; never persist a partial.
- **ISOLATION:** additive only. Do NOT modify `assembleCompleteProfile`, the P3 scoring engine, the generic 360, or the P2 take/submit flow. The only schema change is the NEW `vocational_integrated_results` table; no existing table touched. No prod DDL (dev `prisma db push`; prod via Fargate later).
- All test files live at `api/src/__tests__/*.test.ts` (repo flat convention). Backend test cmds from `api/`. Branch `feat/vocational-360-p4a` (off develop). Commit per task.

import { band, round2 } from "./vocationalScoringService.js";

export interface IntegrationWeights { threeSixty: number; pca: number; mil: number }
export interface IntegrationBands { strong: number; moderateHigh: number; medium: number }
export interface IntegrationConfig {
  instrumentVersion: string;
  integrationWeights: IntegrationWeights;
  bands: IntegrationBands;
}
export interface IntegrationInputs {
  threeSixty: number | null;
  pcaScore: number | null;
  milScore: number | null;
}
export interface IntegratedResultPayload {
  status: "ready";
  instrumentVersion: string;
  integratedComposite: number;
  band: string;
  threeSixtyScore: number;
  pcaScore: number;
  milScore: number;
  weightsApplied: IntegrationWeights;
}
export type IntegrationOutcome = IntegratedResultPayload | { status: "not_ready"; missing: string[] };

export function competencesToScore(competences: { name: string; level: number }[]): number | null {
  if (!competences.length) return null;
  const mean = competences.reduce((s, c) => s + c.level, 0) / competences.length;
  return round2(((mean - 1) / 3) * 100);
}

export function milToScore(mil: {
  milReasoning: number; milDetection: number; milNumeric: number; milMemory: number; milOrientation: number;
}): number {
  const vals = [mil.milReasoning, mil.milDetection, mil.milNumeric, mil.milMemory, mil.milOrientation];
  return round2(vals.reduce((s, v) => s + v, 0) / vals.length);
}

export function normalizeIntegrationWeights(w: IntegrationWeights): IntegrationWeights {
  const sum = w.threeSixty + w.pca + w.mil;
  if (sum === 0) return { threeSixty: 0, pca: 0, mil: 0 };
  return { threeSixty: w.threeSixty / sum, pca: w.pca / sum, mil: w.mil / sum };
}

export function integrate(threeSixty: number, pca: number, mil: number, w: IntegrationWeights): number {
  return round2(threeSixty * w.threeSixty + pca * w.pca + mil * w.mil);
}

export function computeIntegratedResult(config: IntegrationConfig, inputs: IntegrationInputs): IntegrationOutcome {
  const missing: string[] = [];
  if (inputs.threeSixty === null) missing.push("360");
  if (inputs.pcaScore === null) missing.push("pca");
  if (inputs.milScore === null) missing.push("mil");
  if (missing.length) return { status: "not_ready", missing };

  const w = normalizeIntegrationWeights(config.integrationWeights);
  const composite = integrate(inputs.threeSixty as number, inputs.pcaScore as number, inputs.milScore as number, w);
  return {
    status: "ready",
    instrumentVersion: config.instrumentVersion,
    integratedComposite: composite,
    band: band(composite, config.bands),
    threeSixtyScore: inputs.threeSixty as number,
    pcaScore: inputs.pcaScore as number,
    milScore: inputs.milScore as number,
    weightsApplied: w,
  };
}
```

// add to the existing imports at the top of the file:
import {
  computeIntegratedResult, competencesToScore, milToScore,
  type IntegrationConfig, type IntegrationInputs, type IntegrationOutcome,
} from "./vocationalIntegrationService.js";
import { assembleCompleteProfile } from "../lib/assessmentProfile.js";

// ---- P4a integration ----

function asIntegrationWeights(v: unknown): { threeSixty: number; pca: number; mil: number } {
  const o = (v ?? {}) as Record<string, unknown>;
  return {
    threeSixty: typeof o.threeSixty === "number" ? o.threeSixty : 0,
    pca: typeof o.pca === "number" ? o.pca : 0,
    mil: typeof o.mil === "number" ? o.mil : 0,
  };
}

async function loadIntegrationInputs(evaluatedUserId: string): Promise<
  | { ok: false }
  | { ok: true; config: IntegrationConfig; inputs: IntegrationInputs }
> {
  const inst = await prisma.vocationalInstrument.findFirst({
    where: { status: "active", isActive: true },
    select: { version: true, integrationWeights: true, interpretationBands: true },
  });
  if (!inst) return { ok: false };

  const [threeSixtyRes, profile] = await Promise.all([
    getVocationalResult(evaluatedUserId),
    assembleCompleteProfile(evaluatedUserId),
  ]);

  const config: IntegrationConfig = {
    instrumentVersion: inst.version,
    integrationWeights: asIntegrationWeights(inst.integrationWeights),
    bands: asBands(inst.interpretationBands),
  };
  const inputs: IntegrationInputs = {
    threeSixty: threeSixtyRes.status === "ready" ? threeSixtyRes.composite : null,
    pcaScore: competencesToScore(profile.pca.competences ?? []),
    milScore: profile.completeness.lia ? milToScore(profile.lia.mil) : null,
  };
  return { ok: true, config, inputs };
}

export async function recomputeIntegratedResult(
  evaluatedUserId: string,
): Promise<IntegrationOutcome | { status: "never_computed" }> {
  const loaded = await loadIntegrationInputs(evaluatedUserId);
  if (!loaded.ok) return { status: "never_computed" };

  const outcome = computeIntegratedResult(loaded.config, loaded.inputs);
  if (outcome.status !== "ready") return outcome;

  await prisma.vocationalIntegratedResult.upsert({
    where: { evaluatedUserId_instrumentVersion: { evaluatedUserId, instrumentVersion: outcome.instrumentVersion } },
    update: {
      integratedComposite: outcome.integratedComposite, band: outcome.band,
      threeSixtyScore: outcome.threeSixtyScore, pcaScore: outcome.pcaScore, milScore: outcome.milScore,
      weightsApplied: outcome.weightsApplied as object,
    },
    create: {
      evaluatedUserId, instrumentVersion: outcome.instrumentVersion,
      integratedComposite: outcome.integratedComposite, band: outcome.band,
      threeSixtyScore: outcome.threeSixtyScore, pcaScore: outcome.pcaScore, milScore: outcome.milScore,
      weightsApplied: outcome.weightsApplied as object,
    },
  });
  return outcome;
}

export interface SerializedIntegratedResult {
  status: "ready";
  evaluatedUserId: string;
  instrumentVersion: string;
  integratedComposite: number;
  band: string;
  threeSixtyScore: number;
  pcaScore: number;
  milScore: number;
  weightsApplied: unknown;
  computedAt: Date;
}

export function serializeIntegratedResult(row: {
  evaluatedUserId: string; instrumentVersion: string; integratedComposite: unknown; band: string;
  threeSixtyScore: unknown; pcaScore: unknown; milScore: unknown; weightsApplied: unknown; computedAt: Date;
}): SerializedIntegratedResult {
  return {
    status: "ready",
    evaluatedUserId: row.evaluatedUserId,
    instrumentVersion: row.instrumentVersion,
    integratedComposite: Number(row.integratedComposite),
    band: row.band,
    threeSixtyScore: Number(row.threeSixtyScore),
    pcaScore: Number(row.pcaScore),
    milScore: Number(row.milScore),
    weightsApplied: row.weightsApplied,
    computedAt: row.computedAt,
  };
}

export async function getIntegratedResult(
  evaluatedUserId: string,
): Promise<SerializedIntegratedResult | { status: "never_computed" }> {
  const inst = await prisma.vocationalInstrument.findFirst({
    where: { status: "active", isActive: true },
    select: { version: true },
  });
  const version = inst?.version;
  const row = version
    ? await prisma.vocationalIntegratedResult.findUnique({
        where: { evaluatedUserId_instrumentVersion: { evaluatedUserId, instrumentVersion: version } },
      })
    : null;
  if (!row) return { status: "never_computed" };
  return serializeIntegratedResult(row);
}
```
